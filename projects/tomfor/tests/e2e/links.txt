# Links-driven acceptance cases for the group map app (/colony/mareri).
# Format: <url> # <id> expect=<overlay|error> <free note>
# Query params (lat, lon, min, modes, lang, time) are still parsed by the runner for other apps.
# Relative URLs are resolved against BASE (default https://frontend1.torget.bjarby.com).
# mareri reads no query params (only ?stub=1); state lives in the share hash. time and lang are not in the hash,
# so the old ?lat=&lon=&min=&modes=&lang=&time= links are rewritten as hash links without time/lang.
# Swedish UI is covered by acceptance R15 and pr26 share-link-lang instead.
/colony/mareri/#ll=59.33130,18.05930&t=30&w=7&x=3&d=10&r=5&m=bus,metro,tram,train,commuter_rail,boat # weekday-30 expect=overlay T-Centralen, 30 min, all modes
/colony/mareri/#ll=59.33130,18.05930&t=30&w=7&x=3&d=10&r=5&m=metro,bus # metro-bus-30 expect=overlay T-Centralen, metro+bus
/colony/mareri/#ll=59.33130,18.05930&t=120&w=7&x=3&d=10&r=5&m=bus,metro,tram,train,commuter_rail,boat # long-120 expect=overlay 120 min, 24 bands
/colony/mareri/#ll=59.20000,19.20000&t=30&w=7&x=3&d=10&r=5&m=bus,metro,tram,train,commuter_rail,boat # open-water expect=error Baltic sea, no stops near origin
# mareri PR 26 share links: state in the URL hash (ll=lat,lon t=max travel w=max wait x=max transfers d/r=walk dest/transfer m=modes).
# needs=hash: SKIP while the app ignores the hash. lang is not in PR 26's hash (it lives in localStorage), so no lang link here.
/colony/mareri/#ll=59.33260,18.06490&t=45&w=10&x=2&d=8&r=4&m=bus,metro # share-hash-45 expect=overlay needs=hash share link, 45 min, bus+metro
/colony/mareri/#ll=59.20000,19.20000&t=30&w=7&x=3&d=10&r=5&m=bus,metro,tram,train,commuter_rail,boat # share-hash-water expect=error needs=hash share link into the Baltic
